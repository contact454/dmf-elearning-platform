**Features:**
- `RateLimiter` class (in-memory, 100 req/min per IP)
- `checkRateLimit(request)` - Rate limit enforcement
- `addSecurityHeaders(response)` - Security headers injection
- `addCORSHeaders(response, request)` - CORS configuration
- `createSecureResponse(data, request)` - Secure JSON response
- `createSecureErrorResponse(msg, status, request)` - Secure error response

### Pattern Applied to All Routes
```typescript
// BEFORE (insecure)
export async function GET(request: NextRequest) {
  return NextResponse.json(data);
}

// AFTER (secure)
import { withAuth } from '@/middleware/auth';
import { checkRateLimit, createSecureResponse } from '@/middleware/security';

export const GET = withAuth(async (request: NextRequest, { user }) => {
  checkRateLimit(request);
  const userId = user.userId; // From JWT
  // ... business logic
  return createSecureResponse(data, request);
});
```
