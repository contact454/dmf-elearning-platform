### Gotchas

#### ⚠️ Gotcha 1: UUID Type Mismatch
**Problem:** Prisma generates `string` type for UUIDs, but PostgreSQL expects `uuid` type.

**Solution:** Always use `@db.Uuid` annotation:
```prisma
id String @id @default(uuid()) @db.Uuid
```

**Why:** Without `@db.Uuid`, Prisma uses `text` type → slower queries, larger indexes.

---

#### ⚠️ Gotcha 2: JSONB Column Queries
**Problem:** Querying JSONB columns in Prisma requires raw SQL.

**Example (DON'T DO THIS):**
```typescript
// ❌ Doesn't work - Prisma doesn't support JSONB path queries
const exercises = await prisma.readingExercise.findMany({
  where: {
    exerciseData: {
      path: ['correct_index'],
      equals: 0
    }
  }
});
```

**Solution (DO THIS):**
```typescript
// ✅ Use raw query for JSONB filtering
const exercises = await prisma.$queryRaw`
  SELECT * FROM reading_exercises
  WHERE exercise_data->>'correct_index' = '0'
`;
```

**Alternative:** Denormalize frequently-queried JSONB fields into columns.

---

#### ⚠️ Gotcha 3: Decimal Type in TypeScript
**Problem:** Prisma returns `Decimal` as `Prisma.Decimal` object, not `number`.

**Example:**
```typescript
const progress = await prisma.userReadingProgress.findUnique({ ... });
console.log(progress.accuracyPercentage); // Prisma.Decimal object, not 92.5
```

**Solution:**
```typescript
// Convert to number
const accuracyNumber = progress.accuracyPercentage.toNumber();

// Or use in calculations
const avgAccuracy = Prisma.Decimal.sum(accuracyValues).div(accuracyValues.length);
```

```typescript
import { validateMultipleChoice } from './multiple-choice';
import { validateTrueFalse } from './true-false';
import { validateFillBlank } from './fill-blank';
import { validateSequencing } from './sequencing';

export interface ValidationResult {
  isCorrect: boolean;
  accuracyScore: number;
}

export function validateExerciseAnswer(
  exerciseType: string,
  exerciseData: any,
  userAnswer: any
): ValidationResult {
  switch (exerciseType) {
    case 'multiple_choice':
      return validateMultipleChoice(exerciseData, userAnswer);
    case 'true_false':
      return validateTrueFalse(exerciseData, userAnswer);
    case 'fill_blank':
      return validateFillBlank(exerciseData, userAnswer);
    case 'sequencing':
      return validateSequencing(exerciseData, userAnswer);
    default:
      throw new Error(`Unknown exercise type: ${exerciseType}`);
  }
}
```

**File:** `services/reading-command-service/src/domain/validation/fill-blank.ts`

```typescript
import { levenshteinSimilarity } from '@dmf/shared/lib/validation/levenshtein';
import { ValidationResult } from './index';

interface FillBlankData {
  correct_answer: string;
  alternatives?: string[];
}

interface FillBlankAnswer {
  answer: string;
}

export function validateFillBlank(
  exerciseData: FillBlankData,
  userAnswer: FillBlankAnswer
): ValidationResult {
  const userAnswerLower = userAnswer.answer.trim().toLowerCase();
  const correctAnswerLower = exerciseData.correct_answer.toLowerCase();
  
  // Exact match
  if (userAnswerLower === correctAnswerLower) {
    return { isCorrect: true, accuracyScore: 100 };
  }
  
  // Check alternatives
  if (exerciseData.alternatives) {
    const altMatch = exerciseData.alternatives.some(
      alt => alt.toLowerCase() === userAnswerLower
    );
    if (altMatch) {
      return { isCorrect: true, accuracyScore: 100 };
    }
  }
  
  // Fuzzy match (85% threshold)
  const similarity = levenshteinSimilarity(userAnswerLower, correctAnswerLower);
  const accuracyScore = Math.round(similarity * 100);
  
  return {
    isCorrect: similarity >= 0.85,
    accuracyScore
  };
}
```

**File:** `services/shared/lib/validation/levenshtein.ts`

```typescript
export function levenshteinDistance(str1: string, str2: string): number {
  const m = str1.length;
  const n = str2.length;
  
  // Create 2D array (m+1 x n+1)
  const dp: number[][] = Array(m + 1)
    .fill(null)
    .map(() => Array(n + 1).fill(0));
  
  // Initialize first row and column
  for (let i = 0; i <= m; i++) dp[i][0] = i;
  for (let j = 0; j <= n; j++) dp[0][j] = j;
  
  // Fill DP table
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (str1[i - 1] === str2[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1]; // No operation needed
      } else {
        dp[i][j] = 1 + Math.min(
          dp[i - 1][j],     // Deletion
          dp[i][j - 1],     // Insertion
          dp[i - 1][j - 1]  // Substitution
        );
      }
    }
  }
  
  return dp[m][n];
}

export function levenshteinSimilarity(str1: string, str2: string): number {
  const maxLen = Math.max(str1.length, str2.length);
  if (maxLen === 0) return 1.0; // Both empty strings
  
  const distance = levenshteinDistance(str1, str2);
  return (maxLen - distance) / maxLen;
}

// Unit tests (vitest)
if (import.meta.vitest) {
  const { describe, it, expect } = import.meta.vitest;
  
  describe('levenshteinSimilarity', () => {
    it('exact match returns 1.0', () => {
      expect(levenshteinSimilarity('fox', 'fox')).toBe(1.0);
    });
    
    it('case-insensitive match', () => {
      expect(levenshteinSimilarity('fox', 'FOX')).toBe(1.0); // After toLowerCase
    });
    
    it('one character difference', () => {
      expect(levenshteinSimilarity('fox', 'fix')).toBeCloseTo(0.667, 2);
    });
    
    it('accepts 85%+ similarity', () => {
      const similarity = levenshteinSimilarity('running', 'runing'); // 1 char missing
      expect(similarity).toBeGreaterThanOrEqual(0.85); // 6/7 = 0.857
    });
  });
}
```

### API Route Example (Fastify)

**File:** `services/reading-command-service/src/routes/submit.route.ts`

```typescript
import { FastifyInstance } from 'fastify';
import { z } from 'zod';
import { prisma } from '@dmf/shared/lib/prisma';
import { validateExerciseAnswer } from '../domain/validation';
import { emitEvent } from '@dmf/shared/lib/events';

const SubmitRequestSchema = z.object({
  passageId: z.string().uuid(),
  exerciseId: z.string().uuid(),
  userAnswer: z.any(), // Validated per exercise type
  timeSpentSeconds: z.number().int().min(0).max(3600)
});

export async function submitRoutes(fastify: FastifyInstance) {
  fastify.post('/api/reading/submit', {
    preHandler: [fastify.authenticate], // JWT validation middleware
    schema: {
      body: SubmitRequestSchema,
      response: {
        200: {
          type: 'object',
          properties: {
            attemptId: { type: 'string' },
            isCorrect: { type: 'boolean' },
            accuracyScore: { type: 'number' },
            correctAnswer: { type: 'object' },
            explanation: { type: 'string' }
          }
        }
      }
    }
  }, async (request, reply) => {
    const { passageId, exerciseId, userAnswer, timeSpentSeconds } = request.body;
    const userId = request.user.id; // From JWT
    
    // Fetch exercise
    const exercise = await prisma.readingExercise.findUnique({
      where: { id: exerciseId }
    });
    
    if (!exercise) {
      return reply.status(404).send({ error: 'Exercise not found' });
    }
    
    // Validate answer
    const validation = validateExerciseAnswer(
      exercise.exerciseType,
      exercise.exerciseData,
      userAnswer
    );
    
    // Save attempt
    const attempt = await prisma.readingAttempt.create({
      data: {
        userId,
        passageId,
        exerciseId,
        userAnswer,
        correctAnswer: exercise.exerciseData,
        isCorrect: validation.isCorrect,
        accuracyScore: validation.accuracyScore,
        timeSpentSeconds
      }
    });
    
    // Emit event (for progress tracking)
    await emitEvent('ReadingExerciseCompleted', {
      userId,
      passageId,
      exerciseId,
      isCorrect: validation.isCorrect,
      accuracyScore: validation.accuracyScore,
      timeSpentSeconds
    });
    
    return {
      attemptId: attempt.id,
      isCorrect: validation.isCorrect,
      accuracyScore: validation.accuracyScore.toNumber(),
      correctAnswer: exercise.exerciseData,
      explanation: exercise.explanation
    };
  });
}
```

### Gotchas

#### ⚠️ Gotcha 4: Fastify Schema Validation Conflicts with Zod
**Problem:** Fastify's built-in schema validation and Zod both validate, causing redundant work.

**Solution:** Use only Zod, disable Fastify schema:
```typescript
fastify.post('/api/reading/submit', {
  preHandler: [fastify.authenticate],
  // No schema key
}, async (request, reply) => {
  // Manual Zod validation
  const body = SubmitRequestSchema.parse(request.body);
  // ...
});
```

**Or:** Use `fastify-zod` plugin (recommended):
```typescript
import fastifyZod from 'fastify-zod';
fastify.register(fastifyZod);

fastify.post('/api/reading/submit', {
  schema: { body: SubmitRequestSchema }
}, async (request, reply) => {
  // Auto-validated by plugin
});
```

---

#### ⚠️ Gotcha 5: Event Emission Failures Block API Response
**Problem:** If `emitEvent` throws, API returns 500 (even though attempt was saved).

**Solution:** Use fire-and-forget pattern:
```typescript
// ✅ Don't await event emission
emitEvent('ReadingExerciseCompleted', { ... })
  .catch(err => logger.error('Event emission failed', err));

// Return response immediately
return { attemptId, isCorrect, ... };
```

**Why:** Events are async side effects; API response shouldn't depend on them.

```typescript
import { useQuery } from '@tanstack/react-query';

interface Passage {
  id: string;
  title: string;
  cefrLevel: string;
  topic: string;
  wordCount: number;
  estimatedReadingTimeMinutes: number;
  difficultyScore: number;
  isPremium: boolean;

interface UseReadingPassagesOptions {
  cefr?: string;
  topic?: string;
  page?: number;
  limit?: number;
}

export function useReadingPassages(options: UseReadingPassagesOptions = {}) {
  const { cefr, topic, page = 1, limit = 10 } = options;
  
  return useQuery({
    queryKey: ['reading-passages', cefr, topic, page, limit],
    queryFn: async () => {
      const params = new URLSearchParams();
      if (cefr) params.set('cefr', cefr);
      if (topic) params.set('topic', topic);
      params.set('page', page.toString());
      params.set('limit', limit.toString());
      
      const res = await fetch(`/api/reading/passages?${params}`);
      if (!res.ok) throw new Error('Failed to fetch passages');
      
      return res.json() as Promise<{
        passages: Passage[];
        pagination: {
          page: number;
          limit: number;
          total: number;
          totalPages: number;
        };
      }>;
    }
  });
}
```

### Gotchas

#### ⚠️ Gotcha 6: React Query Cache Invalidation
**Problem:** After submitting exercise, progress stats don't update until manual refresh.

**Solution:** Invalidate cache after mutation:
```typescript
const submitMutation = useMutation({
  mutationFn: async (data) => {
    const res = await fetch('/api/reading/submit', {
      method: 'POST',
      body: JSON.stringify(data)
    });
    return res.json();
  },
  onSuccess: () => {
    // ✅ Invalidate progress cache
    queryClient.invalidateQueries({ queryKey: ['reading-progress'] });
    queryClient.invalidateQueries({ queryKey: ['reading-passages'] }); // If passage status changes
  }
});
```

---

#### ⚠️ Gotcha 7: Optimistic Updates Cause UI Flicker
**Problem:** Optimistic update shows correct answer, then API returns error → UI flickers.

**Solution:** Use pessimistic updates for critical actions:
```typescript
// ❌ Optimistic (risky for exercise submission)
const submitMutation = useMutation({
  mutationFn: submitExercise,
  onMutate: async (newAttempt) => {
    // Show feedback immediately (risky if API fails)
    setShowFeedback(true);
  }
});

// ✅ Pessimistic (wait for API response)
const submitMutation = useMutation({
  mutationFn: submitExercise,
  onSuccess: (data) => {
    // Show feedback only after API confirms
    setShowFeedback(true);
    setIsCorrect(data.isCorrect);
  }
});
```

**When to use optimistic updates:**
- Low-risk actions (like buttons, save vocabulary)
- Actions that rarely fail (save word, toggle bookmark)

**When to use pessimistic updates:**
- Critical actions (submit exercise, payment)
- Actions with validation (answer checking)

### Logging Best Practices

**Structured Logging (Pino):**
```typescript
import pino from 'pino';

const logger = pino({
  level: process.env.LOG_LEVEL || 'info',
  formatters: {
    level(label) {
      return { level: label };
    }
  }
});

// Good log (structured)
logger.info({
  event: 'exercise_submitted',
  userId: 'uuid',
  passageId: 'uuid',
  exerciseId: 'uuid',
  isCorrect: true,
  accuracyScore: 100,
  timeSpentSeconds: 45
});

// Bad log (unstructured)
logger.info(`User ${userId} submitted exercise ${exerciseId} - correct!`);
```

**Why:** Structured logs enable easy filtering/aggregation in log analysis tools (Datadog, CloudWatch).

## 🛡️ Security Best Practices

### Input Sanitization

**Sanitize User-Generated Content:**
```typescript
import DOMPurify from 'isomorphic-dompurify';

// If users can add notes to passages (future feature)
const sanitizedNote = DOMPurify.sanitize(userInput, {
  ALLOWED_TAGS: ['b', 'i', 'em', 'strong', 'u'],
  ALLOWED_ATTR: []
});
```

**Why:** Prevents XSS attacks via user-generated content.

### Rate Limiting Per User

**Why:** Prevent single user from exhausting database connections.

```typescript
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(10, '10 s'),
  analytics: true,
  prefix: 'reading:submit'
});

// In route handler
const { success, limit, reset, remaining } = await ratelimit.limit(userId);

if (!success) {
  return reply.status(429).send({
    error: 'Too many requests',
    retryAfter: Math.ceil((reset - Date.now()) / 1000)
  });
}
```
